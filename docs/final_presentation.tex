\begin{frame}{Problem Statement}
\sectiontag{The Challenge}

\textbf{\color{Primary}\normalsize Pharmaceutical Last-Mile Delivery}

\vspace{0.05cm}
\footnotesize
Pharmaceutical delivery operations involve multiple destinations, shipment
requirements and time constraints. Efficient planning becomes difficult when
travel conditions and operational resources vary.

\vspace{0.15cm}

\begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.52\textwidth}
        \textbf{\color{Primary}\small Key operational challenges addressed}

        \begin{itemize}
            \setlength{\itemsep}{0pt}
            \item Multiple delivery locations and varying shipment demands
            \item Dual knapsack limits: cargo weight and volume capacities
            \item Strict clinic time windows and unloading service times
            \item Driver shift limits and labor regulations
            \item Urban traffic uncertainty across road networks
            \item Balancing route efficiency with on-time delivery quality
        \end{itemize}
    \end{column}

    \begin{column}{0.45\textwidth}
        \textbf{\color{Primary}\small Core problem}

        \vspace{0.04cm}
        \footnotesize
        How can pharmaceutical deliveries be planned efficiently while
        satisfying operational constraints?

        \vspace{0.2cm}
        \textbf{\color{Primary}\small What we developed}

        \vspace{0.04cm}
        \footnotesize
        An interactive decision-support system using Operations
        Research to generate, evaluate, and stress-test feasible delivery plans in real time.
    \end{column}
\end{columns}
\end{frame}

\begin{frame}{Mathematical Model Formulation}

\sectiontag{Mathematical Formulation}

\vspace{0.05cm}

\small

\begin{columns}[T,totalwidth=\textwidth]


% LEFT COLUMN
\begin{column}{0.48\textwidth}

\textbf{\color{Primary}\small Decision Variables}

\vspace{0.05cm}

\[
\scriptsize
x_{ijk} \in \{0, 1\}
\]
\footnotesize
Binary routing variable: $1$ if vehicle $k$ travels directly from $i$ to $j$; $0$ otherwise.

\vspace{0.05cm}

\[
\scriptsize
y_{ik} \in \{0, 1\}
\]
\footnotesize
Assignment variable: $1$ if vehicle $k$ serves customer $i$; $0$ otherwise.

\vspace{0.05cm}

\[
\scriptsize
T_i \ge 0, \quad L_i \ge 0
\]
\footnotesize
Arrival time $T_i$ and delivery lateness penalty $L_i = \max(0, T_i - l_i)$.

\end{column}


% RIGHT COLUMN
\begin{column}{0.48\textwidth}

\textbf{\color{Primary}\small Objective Function}

\vspace{0.05cm}

\[
\scriptsize
\min Z = \alpha \left(\frac{D}{D_0}\right) + \beta \left(\frac{T}{T_0}\right) + \gamma \left(\frac{L}{L_0}\right)
\]

\vspace{-0.05cm}

\footnotesize
where:
\begin{itemize}
\setlength\itemsep{1pt}
\scriptsize
\item $D$ = Total fleet travel distance
\item $T$ = Total delivery transit time
\item $L$ = Total delivery lateness penalty
\item $\alpha,\beta,\gamma$ = Importance priority weights
\end{itemize}

\vspace{0.05cm}

\textbf{\color{Primary}\small Constraints (in plain terms)}

\begin{itemize}
\footnotesize
\setlength\itemsep{1pt}
\item Every delivery request is served exactly once
\item Vehicle weight and volume limits are not exceeded
\item Delivery occurs within customer time windows
\item Driver shift duration cutoff is respected
\end{itemize}

\end{column}

\end{columns}

\end{frame}

% ------------------------------------------------------------
% 10. WHY DECOMPOSED MILP (DMILP)?
% ------------------------------------------------------------

\begin{frame}{Why Decomposed MILP (DMILP)?}
\sectiontag{Algorithmic Rationale}

\vspace{-0.15cm}

\begin{columns}[T,totalwidth=\textwidth]

    \begin{column}{0.48\textwidth}
        \textbf{\color{Primary}\small The Optimization Dilemma}
        \vspace{0.05cm}
        \footnotesize
        \begin{itemize}
            \setlength{\itemsep}{4pt}
            \item \textbf{Why not Monolithic MILP alone?}
            For 60--85 delivery locations, a single MILP formulation generates $>7,000$ binary variables and $>8,500$ constraints. Exact commercial/open-source solvers timeout ($>300\text{ seconds}$), making it impossible to support interactive planning.
            \item \textbf{Why not pure Heuristics alone?}
            Greedy heuristics run fast ($<5\text{ ms}$), but they lack mathematical rigor, can produce suboptimal route sequences, and struggle to pack multiple tight constraints simultaneously.
        \end{itemize}
    \end{column}
