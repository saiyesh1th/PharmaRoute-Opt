\begin{frame}{Mathematical Model: Multi-Objective CVRPTW}
\sectiontag{Mixed-Integer Linear Programming Formulation}

\vspace{-0.15cm}

\begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.48\textwidth}
        \textbf{\color{Primary}\small Decision Variables}
        \vspace{0.04cm}
        \begin{itemize}
            \setlength{\itemsep}{2pt}
            \scriptsize
            \item $x_{ijk} \in \{0, 1\}$: Binary arc traversal from node $i$ to $j$ by vehicle $k$.
            \item $y_{ik} \in \{0, 1\}$: Assignment of customer $i$ to vehicle $k$.
            \item $T_{ik} \ge 0$: Continuous arrival timestamp at customer $i$.
            \item $L_i \ge 0$: Soft deadline lateness penalty ($\max(0, T_i - l_i)$).
        \end{itemize}

        \vspace{0.15cm}
        \textbf{\color{Primary}\small Multi-Objective Goal Function}
        \vspace{0.04cm}
        \[
        \scriptsize
        \min Z = \alpha \left(\frac{D}{D_0}\right) + \beta \left(\frac{T}{T_0}\right) + \gamma \left(\frac{L}{L_0}\right)
        \]
        \tiny\color{Muted}
        Normalized weighted sum: $D_0, T_0, L_0$ are scalar ideal bounds; $\alpha + \beta + \gamma = 1.0$.
    \end{column}

    \begin{column}{0.50\textwidth}
        \textbf{\color{Primary}\small Operational Constraints}
        \vspace{0.04cm}
        \scriptsize
        \begin{align*}
            & \sum_{k \in \mathcal{K}} y_{ik} = 1 \quad \forall i \in \mathcal{V}_c & \text{(Coverage)} \\
            & \sum_{j} x_{ijk} = \sum_{j} x_{jik} = y_{ik} & \text{(Flow Conservation)} \\
            & \sum_{i \in \mathcal{V}_c} w_i y_{ik} \le C_w, \; \sum_{i \in \mathcal{V}_c} v_i y_{ik} \le C_v & \text{(Dual Knapsack)} \\
            & T_{jk} \ge T_{ik} + s_i + t_{ij} - M(1 - x_{ijk}) & \text{(MTZ Subtour)} \\
            & T_{ik} \ge e_i y_{ik}, \quad T_{ik} \le l_i + L_i + M(1 - y_{ik}) & \text{(Time Windows)} \\
            & T_{ik} + s_i + t_{i0} \le T_{\max} + M(1 - x_{i0k}) & \text{(Shift Duration)}
        \end{align*}
    \end{column}
\end{columns}

\vfill
\end{frame}

\begin{frame}{Algorithmic Methodology: Overcoming the Scale Bottleneck}
\sectiontag{Cluster-First Route-Second Decomposition}

\vspace{-0.15cm}

\begin{columns}[T,totalwidth=\textwidth]
    \begin{column}{0.48\textwidth}
        \textbf{\color{Primary}\small The Scalability Bottleneck}
        \vspace{0.04cm}
        \footnotesize
        \begin{itemize}
            \setlength{\itemsep}{3pt}
            \item CVRPTW with MTZ formulation is strongly $\mathcal{NP}$-hard.
            \item On 78-clinic instances, monolithic MILP generates $>7,000$ binary variables and $>8,500$ constraints.
            \item Result: Monolithic solvers time out ($>300\text{ s}$), failing to find proven integer solutions.
        \end{itemize}

        \vspace{0.2cm}
        \begin{tikzpicture}
            \node[fill=SoftBlue, draw=Line, rounded corners=3pt, font=\scriptsize, inner sep=4pt, text width=0.96\linewidth] {
                \textbf{\color{Primary}Our Solution:} Cluster-First Route-Second Decomposition with exact per-cluster MILP sub-solvers.
            };
        \end{tikzpicture}
    \end{column}

    \begin{column}{0.50\textwidth}
        \textbf{\color{Primary}\small Three-Stage Decomposition Pipeline}
        \vspace{0.06cm}

        \begin{beamercolorbox}[rounded=true,shadow=false,wd=\linewidth,sep=4pt]{block body}
            {\bfseries\color{Primary}\scriptsize Stage 1: Medoid Dispersion Clustering}\par\vspace{0.02cm}
            \scriptsize
            Max-min dispersion medoid seeding under dual knapsack constraints and time-window urgency sorting.
        \end{beamercolorbox}

        \vspace{0.1cm}

        \begin{beamercolorbox}[rounded=true,shadow=false,wd=\linewidth,sep=4pt]{block body}
            {\bfseries\color{Accent}\scriptsize Stage 2: HiGHS Branch-and-Cut Sub-MIPs}\par\vspace{0.02cm}

        \end{beamercolorbox}
    \end{column}
